```javascript
const conversationsRef = useRef(conversations);
useEffect(() => {
    conversationsRef.current = conversations;

const openConversation = useCallback(
    async (id) => {
        setCurrentConversationId(id);
        const chat = conversationsRef.current.find((c) => String(c.id) === String(id));
        if (!chat) return;

        const sessionToken = activeSessionRef.current;
        const currentUserId = activeUserIdRef.current;

        if (chat.loaded === false) {
            try {
                const data = await apiGetConversation(id);
                if (activeSessionRef.current !== sessionToken || activeUserIdRef.current !== currentUserId) {
                    return;
                }
                const loadedMessages = (data.messages || []).map((m) => ({
                    sender: m.sender === 'assistant' ? 'ai' : 'user',
                    text: m.text
                }));
                setConversations((prev) => {
                    if (activeSessionRef.current !== sessionToken || activeUserIdRef.current !== currentUserId) {
                        return [];
                    }
                    return prev.map((c) =>
                        String(c.id) === String(id)
                            ? { ...c, messages: loadedMessages, loaded: true }
                            : c
                    );
                });
            } catch (err) {
                console.error('Failed to load conversation history:', err);
            }
        }
    },
    [] // Stable callback — does NOT re-create when conversations change
);
```

```javascript
const logout = useCallback(async (isRemoteBroadcast = false) => {
    // 1. Guard against duplicate/redundant logouts
    if (!currentUser && !localStorage.getItem('guardrailToken')) {
        clearLoggedOutCookie();
        return;
    }

    if (onLogoutRef.current) {
        try {
            onLogoutRef.current();
        } catch (err) {
            console.warn('Error clearing conversation state on logout:', err);
        }
    }

    setCurrentUser(null);
    setAuthToken(null);

    localStorage.removeItem('guardrailUser');
    localStorage.removeItem('guardrailToken');
    localStorage.removeItem('guardrailChats');
    localStorage.removeItem('eclipse_admin_token');
    localStorage.removeItem('eclipse_admin_user');
    clearLoggedOutCookie();

    // 2. Only broadcast to OTHER tabs if this wasn't triggered by an incoming broadcast
    if (!isRemoteBroadcast) {
        try {
            const channel = new BroadcastChannel('eclipse_auth_channel');
            channel.postMessage({ type: 'ECLIPSE_LOGOUT' });
            channel.close();
        } catch {}
    }

    try {
        await apiLogout();
    } catch (err) {
        console.warn('Logout request failed:', err);
    }

    showToast('Logged out.', 'success');
}, [currentUser]);

// Cross-tab and cross-port logout synchronization
useEffect(() => {
    let channel = null;
    try {
        channel = new BroadcastChannel('eclipse_auth_channel');
        channel.onmessage = (e) => {
            if (e.data?.type === 'ECLIPSE_LOGOUT') {
                logout(true); // isRemoteBroadcast = true prevents re-broadcasting
            }
        };
    } catch {}

    const handleSyncCheck = () => {
        if (authToken || currentUser) {
            clearLoggedOutCookie();
            return;
        }
        const cookie = document.cookie || '';
        if (cookie.includes('eclipse_logged_out=1') && (authToken || currentUser)) {
            logout(true);
        }
    };

    window.addEventListener('focus', handleSyncCheck);
    document.addEventListener('visibilitychange', handleSyncCheck);
    window.addEventListener('storage', (e) => {
        if (e.key === 'guardrailToken' && !e.newValue && (authToken || currentUser)) {
            logout(true);
        }
    });

    return () => {
        if (channel) channel.close();
        window.removeEventListener('focus', handleSyncCheck);
        document.removeEventListener('visibilitychange', handleSyncCheck);
    };
}, [logout, authToken, currentUser]);
```

```python
async def get_current_user_id(
    credentials: HTTPAuthorizationCredentials | None = Depends(_bearer_scheme),
) -> str:
    if credentials is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Missing Authorization header. Expected: Bearer <token>",
        )

    payload = decode_access_token(credentials.credentials)
    if payload is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token.",
        )

    user_id = payload.get("sub") or payload.get("user_id") or payload.get("id")
    if not user_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token payload: missing user identity (sub/user_id).",
        )

    return str(user_id)
```

```javascript
const User = require("../models/User");
const AuditLog = require("../models/AuditLog");
const ApiUsage = require("../models/ApiUsage");
const pythonMlClient = require("../services/pythonMlClient");

/**
 * @route   GET /api/admin/users
 * @desc    Get all users with activity counts
 * @access  Private (Admin)
 */
const getUsers = async (req, res) => {
  try {
    const users = await User.find({}).sort({ created_at: -1 }).lean();

    const usersWithStats = await Promise.all(
      users.map(async (u) => {
        const userIdStr = u._id.toString();
        const requestCount = await AuditLog.countDocuments({ user_id: userIdStr });
        return {
          user_id: userIdStr,
          name: u.name,
          email: u.email,
          created_at: u.created_at ? new Date(u.created_at).toISOString() : null,
          is_flagged: Boolean(u.is_flagged),
          is_admin: Boolean(u.is_admin),
          request_count: requestCount,
        };
      })
    );

    return res.json({ users: usersWithStats, count: usersWithStats.length });
  } catch (error) {
    console.error("[Admin getUsers Error]:", error);
    return res.status(500).json({ success: false, detail: "Failed to fetch users." });
  }
};

/**
 * @route   GET /api/admin/stats
 * @desc    System-wide statistics for Admin Dashboard
 * @access  Private (Admin)
 */
const getAdminStats = async (req, res) => {
  try {
    const [
      totalUsers,
      flaggedUsers,
      totalRequests,
      statusAgg,
      apiDocs
    ] = await Promise.all([
      User.countDocuments({}),
      User.countDocuments({ is_flagged: true }),
      AuditLog.countDocuments({}),
      AuditLog.aggregate([{ $group: { _id: "$status", count: { $sum: 1 } } }]),
      ApiUsage.find({}).lean()
    ]);

    // Unique conversations count from AuditLogs
    const uniqueConversations = await AuditLog.distinct("session_id");

    const statusBreakdown = {};
    statusAgg.forEach((item) => {
      if (item._id) statusBreakdown[item._id] = item.count;
    });

    const apiUsage = {};
    apiDocs.forEach((doc) => {
      apiUsage[doc.api_name.toLowerCase()] = {
        requests: doc.requests_count || 0,
        is_rate_limited: Boolean(doc.is_rate_limited),
      };
    });

    // Anomaly count from Python microservice
    let totalAnomalies = 0;
    try {
      const anomalyData = await pythonMlClient.getAnomalies();
      totalAnomalies = (anomalyData.anomalies || []).length;
    } catch {}

    return res.json({
      total_users: totalUsers,
      flagged_users: flaggedUsers,
      total_requests: totalRequests,
      total_conversations: uniqueConversations.length,
      total_anomalies: totalAnomalies,
      status_breakdown: statusBreakdown,
      api_usage: apiUsage,
    });
  } catch (error) {
    console.error("[Admin getAdminStats Error]:", error);
    return res.status(500).json({ success: false, detail: "Failed to compile admin stats." });
  }
};

/**
 * @route   GET /api/admin/anomalies
 * @desc    Fetch detected anomalies via Python ML microservice
 * @access  Private (Admin)
 */
const getAnomalies = async (req, res) => {
  try {
    const data = await pythonMlClient.getAnomalies();
    return res.json(data);
  } catch (error) {
    console.error("[Admin getAnomalies Error]:", error);
    return res.status(500).json({ success: false, detail: "Failed to fetch anomalies." });
  }
};

/**
 * @route   PUT /api/admin/user/:user_id/flag
 * @desc    Toggle user flagged status
 * @access  Private (Admin)
 */
const flagUser = async (req, res) => {
  try {
    const { user_id } = req.params;
    const user = await User.findById(user_id);

    if (!user) {
      return res.status(404).json({ success: false, detail: "User not found." });
    }

    user.is_flagged = !user.is_flagged;
    await user.save();

    return res.json({
      user_id: user._id.toString(),
      is_flagged: user.is_flagged,
    });
  } catch (error) {
    console.error("[Admin flagUser Error]:", error);
    return res.status(500).json({ success: false, detail: "Failed to update user flag status." });
  }
};

module.exports = {
  getUsers,
  getAdminStats,
  getAnomalies,
  flagUser,
};
```

```python
def detect_injection(text: str) -> dict:
    """
    Hybrid Cascade Injection Detector:
    1. Fast-path: Check regex rules. If severe match, block in <1ms.
    2. Deep-path: If regex passes and mode=='ml', run DeBERTa-v3 model.
    """
    # 1. Always evaluate rule heuristics first (catches obvious/known attacks instantly)
    rule_result = _detect_rules(text)
    if rule_result["is_jailbreak"] or rule_result["is_injection"]:
        return rule_result

    # 2. If rules pass and ML mode is active, run DeBERTa-v3 semantic evaluation
    if _MODE == "ml":
        try:
            return _detect_ml(text)
        except Exception as e:
            print(f"[Injection Detector ML Error]: {e} — falling back to rule result")
            return rule_result

    return rule_result
```

TASK 6: MULTI-TIER PRODUCTION CLOUD DEPLOYMENT
================================================================================

Target Production Stack:
- Database: MongoDB Atlas (M0 free or dedicated tier)
- ML Microservice: Hugging Face Spaces (Docker SDK) or Render Web Service
- API Gateway: Render / Railway / AWS ECS (Node.js)
- Frontends: Vercel / Netlify

--------------------------------------------------------------------------------
Step 6.1: Database (MongoDB Atlas)
--------------------------------------------------------------------------------
1. Create a MongoDB Atlas cluster.
2. In Database Access, create user with readWriteAnyDatabase.
3. In Network Access, allow `0.0.0.0/0` (required for serverless/cloud containers).
4. Obtain connection URI: `mongodb+srv://<user>:<password>@cluster0.mongodb.net/llm_guardrail?retryWrites=true&w=majority`.

--------------------------------------------------------------------------------
Step 6.2: Python ML Microservice (Hugging Face Spaces or Render)
--------------------------------------------------------------------------------
1. Set Space SDK to `Docker`.
2. Configure Dockerfile to redirect model caches to `/tmp`:
   `ENV HF_HOME=/tmp/huggingface TRANSFORMERS_CACHE=/tmp/huggingface`
3. Expose port `7860` (HF Spaces default) or `8001` (Render).
4. Set Environment Variables:
   - `MONGO_URI`: Atlas connection string
   - `MONGO_DB_NAME`: `llm_guardrail`
   - `INTERNAL_SERVICE_KEY`: Random 64-char hex string
   - `INJECTION_DETECTOR_MODE`: `ml`
   - `GEMINI_API_KEY`, `GROQ_API_KEY`, etc.

--------------------------------------------------------------------------------
Step 6.3: Node.js Express Gateway (Render / Railway)
--------------------------------------------------------------------------------
1. Set Root Directory to `server/`.
2. Build Command: `npm install`.
3. Start Command: `node server.js`.
4. Set Environment Variables:
   - `PORT`: `5000` (or host provided `$PORT`)
   - `MONGO_URI`: Atlas connection string
   - `JWT_SECRET`: Random 64-char hex string
   - `JWT_EXPIRES_IN`: `7d`
   - `PYTHON_SERVICE_URL`: URL of deployed Python ML service (e.g. `https://your-space.hf.space`)
   - `INTERNAL_SERVICE_KEY`: Matching secret from Step 6.2
   - `CORS_ORIGINS`: Comma-separated URLs of deployed frontends (e.g. `https://eclipse-chat.vercel.app,https://eclipse-admin.vercel.app`)

--------------------------------------------------------------------------------
Step 6.4: Frontends (Vercel / Netlify)
--------------------------------------------------------------------------------
1. User Frontend (`frontend/`):
   - Build Command: `npm run build`
   - Output Directory: `dist`
   - Environment Variable: `VITE_API_BASE=https://your-express-gateway.onrender.com`
2. Admin Frontend (`admin_frontend/`):
   - Build Command: `npm run build`
   - Output Directory: `dist`
   - Environment Variables:
     - `VITE_NODE_API_URL=https://your-express-gateway.onrender.com`
     - `VITE_PYTHON_API_URL=https://your-python-service.hf.space`
